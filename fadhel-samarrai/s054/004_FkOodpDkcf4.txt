"وهذا ممكن، يعني: {لَنَسْفَعًا بِالنَّاصِيَةِ * نَاصِيَةٍ كَاذِبَةٍ خَاطِئَةٍ}، لا يشترط في البدل والمبدل منه التـ... يعني مطابقة التعريف والتنكير، قد يكون المبدل منه معرفة والآخر نكرة، أو العكس، لا يشترط، قد يأتيان متطابقين أو لا يأتيان يعني مختلفين."
"شكراً."
"العفو."
"العفو، مع السلامة، في أمان الله."

2.
"معنا الأخت أم محمد من الشارقة، السلام عليكم."
"السلام عليكم، ممكن أكلم دكتور السامرائي؟"
"الدكتور مع حضرتك."
"إي، لو سمحت، في سورة المؤمنين، في الآية رقم 85: بسم الله الرحمن الرحيم: {سَيَقُولُونَ لِلَّهِ ۚ أَفَلَا تَذَكَّرُونَ}، والآية اللي بعدها: {قُلْ مَن رَّبُّ السَّمَاوَاتِ السَّبْعِ وَرَبُّ الْعَرْشِ الْعَظِيمِ * سَيَقُولُونَ لِلَّهِ}..."
"هسه نجاوب عنها إن شاء الله."
"...{قُلْ أَفَلَا تَتَّقُونَ}."
"طيب."
"واللي بعدها: {قُلْ مَن بِيَدِهِ مَلَكُوتُ كُلِّ شَيْءٍ وَهُوَ يُجِيرُ وَلَا يُجَارُ عَلَيْهِ إِن كُنتُمْ تَعْلَمُونَ * سَيَقُولُونَ لِلَّهِ ۚ قُلْ فَأَنَّىٰ تُسْحَرُونَ}، تُسحرون. أريد أفهم ليش جاءت (لله) بدل من (الله) يعني؟ أنا أعتقد فلابد أنه شيء لا نفهمه نحن في اللغة أو في..."
"طيب، ممكن نجيب عنها."
"إي، شكراً جزيلاً. ونقطة ثانية لو سمحت، نفس الشيء بالنسبة للتشكيل في الآية اللي تقول: {فَاللَّهُ خَيْرٌ حَافِظًا}..."
"نعم."
"...وأيضاً اللي تقول: {سَلَامٌ قَوْلًا مِّن رَّبٍّ رَّحِيمٍ}، التشكيل يعني ضيعني في الإعراب، ما عرفت موقع الكلمات..."
"نعم، {وَكَبُرَتْ كَلِمَةً}..."
"...شكراً جزيلاً."
"العفو، الله يبارك فيكِ."

"تريد هسا نجاوبها؟"
"نعم."
"ولا نرجئها يعني؟"
"نجيب عليها الآن، ولكن بعد هذا الاتصال الهاتفي. أستاذ أشرف من الشارقة، السلام عليكم."
"وعليكم السلام."
"مرحباً بك."
"كل عام وأنتم بخير."
"وأنت بكامل الصحة والسعادة."
"والله عندي سؤال لو سمحت بخصوص الفرق بين فعلي (أتى) و(جاء)، نعم، لأنه تكرر في القرآن مثلاً في الآية الكريمة: {أَتَىٰ أَمْرُ اللَّهِ}، و{إِذَا جَاءَ نَصْرُ اللَّهِ وَالْفَتْحُ}، وفي آية أخرى يأتوا بنفس الآية: {يَا أَبَتِ إِنِّي قَدْ جَاءَنِي مِنَ الْعِلْمِ مَا لَمْ يَأْتِكَ}، نعم، فلو سمحت توضح لنا الفرق بين الاثنين."
"حاضر، في هذه الحلقة إن شاء الله، شكراً لك."

4.
"دكتور فاضل، عود يعني بداية هذا التساؤل الأخت أم محمد: {فَاللَّهُ خَيْرٌ حَافِظًا}، و{سَلَامٌ قَوْلًا}، و{كَبُرَتْ كَلِمَةً}."
"نعم نعم، نعم. يعني {خَيْرٌ حَافِظًا} شوي نحتاج تحتاج إلى شرح، ممكن أذكر..."
"تفضل، تفضل، نعم."
"هو فيها يعني وجهين إعرابيين: إما أن تكون تمييزاً، أو أن تكون حالاً؛ لأن اسم التفضيل، اسم التفضيل إذا كان ما بعده ليس من جنسه..."
"اسم التفضيل هنا {خَيْرٌ}؟"
"نعم، نعم. إذا كان ما بعده ليس من جنسه يُنصب، وإذا كان ما بعده من جنسه يُضاف. تقول: أنت أفضلُ رجلٍ؛ لأن رجل هو أنت أفضل رجل."
"نعم."
"أو: هذه أحسنُ دارٍ، وحديقتك أجملُ حديقةٍ، وهو أحسنُ إنسانٍ، كذا. وإذا كان مغايراً له ينصب، كأن تقول: هو أحسنُ شعراً، يعني فلان أحسن، الشعر ليس من جنس هذا، خالدٌ أحسنُ شعراً، الشعر مغاير للجنس، يسموه فاعل في المعنى، إذا كان فاعلاً في المعنى ينصب: أحسنُ شعراً، يعني وهكذا: أنت مثلاً أفره أو أكثرُ مالاً، المال ليس من جنس أنت، وقالوا: {نَحْنُ أَكْثَرُ أَمْوَالًا}، ينصب، إذا كان من جنسه يُجر."
"امم، {أَنَا أَكْثَرُ مِنكَ مَالًا وَأَعَزُّ نَفَرًا}."
"طبعاً، إذا كان إذا كان من جنسه يجر."
"من غير جنسه ينصب."
"ينصب، يسموه الفاعل في المعنى."
"نعم."
"فـ ممكن نقول: فالله خيرُ حافظٍ، يعني هو الحافظ."
"ممكن."
"ممكن، بس بـ..."
"فالله خيرُ حافظٍ."
"يعني هو الحافظ، نعم، هو الحافظ. {فَاللَّهُ خَيْرٌ حَافِظًا} حفظته أحسن."
"الله!"
"حفظته، الله، ليس هو، وإنما لأن ربنا قال: {وَيُرْسِلُ عَلَيْكُم حَفَظَةً}، فلما تقول: فالله خيرُ حافظٍ هو الحافظ، {خَيْرٌ حَافِظًا} يعني حفظته خير، خير منكم، فهو لم يقارن بينه وبين الله، وإنما بينهم وبين حفظة الله، وهم الملائكة، فالله خير... نعم."
"ولذلك قالها سيدنا يعقوب في يوسف..."
"نعم."
"...نعم، حينما قال أعطاهم أخاه وقال اذهبوا به: {فَاللَّهُ خَيْرٌ حَافِظًا وَهُوَ أَرْحَمُ الرَّاحِمِينَ}."
"فإذن فيها فيها وجهين يعني: التمييز والحالية."
"أيهما أصوب أو أقوى دكتور فاضل؟"
"التمييز."
"التمييز أقوى، نعم."
"لسبب، يعني أخشى إذا بدينا نتكلم فيه يأخذ بقية البرنامج، لسبب يعني القوة، ليست هكذا اعتباطاً يعني."
"نعم، يعني نرجئ لماذا تمييز بالذات للحلقة القادمة؟"
"هنا في في هذه في هذه لماذا هو أقوى؟ لأن يعني إذا تريد نجاوب على السريع أيضاً ممكن، لكن يعني أخشى أن..."
"تفضل."
"تأخذ لنا على تأخذ وقت على الآذان. هو الحافظ يسموها الـ الحال قيد لعاملها."
"نعم."
"الحال قيد لعاملها، يعني شلون؟ يعني مثلاً لما تقول: جاء ضاحكاً، الضحك مقرون بالمجيء."
"امم."
"فعندما تقول تجعلها حال، يعني خير فقط في هذه الحال."
"امم."
"وراح تكون يعني محدودة."
"في حالات الحفظ."
"في حالات الحفظ، بينما ذيك لا، مطلقة، فتكون أقوى."
"ما شاء الله! فتح الله عليك."
"إيه."
"و{سَلَامٌ قَوْلًا}؟"
"هذا مفعول مطلق، يقوله ربنا قولاً، مفعول مطلق لفعل محذوف."
"امم، على غرار {فَتَبَسَّمَ ضَاحِكًا} مثلاً؟"
"ذيك حال."
"حال."
"ذيك حال مؤكدة."
"امم."
"{فَتَبَسَّمَ ضَاحِكًا}؛ لأن هذا اسم فاعل ضاحكاً، بينما هاي {قَوْلًا} مصدر."
"امم."
"{قَوْلًا} مصدر."
"{كَبُرَتْ كَلِمَةً تَخْرُجُ مِنْ أَفْوَاهِهِمْ}؟"
"هذا هذا هذا تمييز، يسموه الفاعل المفسر بتمييز، التمييز ضمير: كبرت الكلمة كلمةً، هذا يسموه هذا تمييز تفسر الفاعل المستتر، الضمير المستتر."
"هذا أصل الكلام خارج القرآن الكريم: كبرت الكلمةُ كلمةً؟"
"هذا هذا الأصل، يعني نحن نستطيع أن نفسر الفاعل بالتمييز."
"نعم."
"ممكن."
"فتح الله على حضرتك. يا أخت أم محمد، لعل تكون الإجابة قد وصلت إن شاء الله. أستاذ دكتور فاضل السامرائي عود إلى ما كنا به بسورة الليل، وربنا تبارك وتعالى قال في سياق آي الذكر الحكيم: ينجي الله، ثم {نُنَجِّي الَّذِينَ اتَّقَوا}، وهنا قال تبارك وتعالى: سيتجنبها..."
"يُجَنَّبُهَا."
"يُجَنَّبُهَا الأتقى، لماذا؟"
"امم، الفرق حقيقة بين التجنيب وبين التنجية: التنجية قد تكون بعد الوقوع في الشيء ومعاناته، يعني يقع في الشيء وتنجيه منه، كقوله تعالى بني يقول: {وَإِذْ نَجَّيْنَاكُم مِّنْ آلِ فِرْعَوْنَ يَسُومُونَكُمْ سُوءَ الْعَذَابِ}..."
"سوء العذاب."
"...يعني إذن هم أُذيقوا العذاب ثم نجاهم منه، كانت النجاة بعد أن وقعوا في المكروه."
"نعم."
"وأيضاً ربنا قال لما ذكر قال: {وَإِن مِّنكُمْ إِلَّا وَارِدُهَا}، قال: {ثُمَّ نُنَجِّي الَّذِينَ اتَّقَوا}، إذن بعد الورود هذا."
"نعم."
"فالـ النجاة قد تكون بعد الورود، يعني بعد حصول المكروه، بعد الوقوع في الشيء."
"الله!"
"ولذا قال: {وَذَا النُّونِ إِذ ذَّهَبَ مُغَاضِبًا فَظَنَّ أَن لَّن نَّقْدِرَ عَلَيْهِ فَنَادَىٰ فِي الظُّلُمَاتِ أَن لَّا إِلَٰهَ إِلَّا أَنتَ سُبْحَانَكَ إِنِّي كُنتُ مِنَ الظَّالِمِينَ * فَاسْتَجَبْنَا لَهُ وَنَجَّيْنَاهُ}..."
"...{وَنَجَّيْنَاهُ} لأنه كان واقعاً، طبعاً، كان واقعاً."
"فالنجـ النجاة قد تكون بعد الوقوع في الشيء، بعد المعاناة النجاة. التجنيب لا، التنحية."
"الله أكبر!"
"فلما هناك قال ذاك ماذا قال في النجاة؟ قال: {ثُمَّ نُنَجِّي الَّذِينَ اتَّقَوا}، هذا الأتقى!"
"هذا الأتقى، وليس يعني هذا الدرجة العليا في التقوى، قال: {يُجَنَّبُهَا}."
"وإذا قال: {الَّذِينَ اتَّقَوا}، والذين اتقوا درجات، أعلاهم الأتقى."
"قال: سيـ..."
"قال: {يُجَنَّبُهَا}، {يُجَنَّبُهَا}."
"يعني لا يقع فيها."
"لا لا يقع لا يقع فيها، تنحية!"
"الله! ما شاء الله!"
"ولذا قال تلاحظ وقال ربنا: {وَيُنَجِّي اللَّهُ الَّذِينَ اتَّقَوْا بِمَفَازَتِهِمْ}، ينجي الله الذين اتقوا، لما يستعمل اتقوا يسـ يستعمل التنجية، لكن الأتقى: تجنيب."
"تنحية فقط."
"التنحية. تحلة القسم هكذا على السريع، يردها تحلة القسم."
"الله أكبر!"
"هكذا، يعني هذه إشارة عظيمة إلى الـ إلى المـ إلى الناس لو كانوا يعني ينتبهون إلى هذا."
"والله لو كانوا يعقلون، إشارة عظيمة، الله أكبر، ما شاء الله. نعم، فتح الله عليك."
"إشارة عظيمة، إشارة عظيمة، نعم."
"بآخر الآيات يقول..."

نعم، بدل هذه، بدل؛ {زِينَةٍ}، {بِزِينَةٍ الْكَوَاكِبِ}، {زِينَةٍ} هكذا، ثم بعدها {الْكَوَاكِبِ} بدل؛ لأن البدل والمبدل منه لا يُشترط تطابقهما تعريفاً وتنكيراً، قد يأتي أحدهما معرفة والآخر نكرة.

نعم.
